from sklearn.metrics import precision_recall_curve, f1_score, jaccard_score

def calculate_metrics(y_true, y_pred, threshold=0.5):
    """Calculate metrics for binary segmentation"""
    y_true_flat = y_true.flatten()
    y_pred_flat = y_pred.flatten()
    
    # Binarize predictions
    y_pred_binary = (y_pred_flat > threshold).astype(float)
    
    # Normalize ground truth
    y_true_binary = (y_true_flat > 127).astype(float)
    
    # Calculate metrics
    iou = jaccard_score(y_true_binary, y_pred_binary)
    f1 = f1_score(y_true_binary, y_pred_binary)
    
    # Precision-recall curve
    precision, recall, thresholds = precision_recall_curve(y_true_binary, y_pred_flat)
    
    # Find precision-recall breakeven point (where precision ≈ recall)
    idx = np.argmin(np.abs(precision - recall))
    breakeven = (precision[idx] + recall[idx]) / 2
    
    return {
        'iou': iou,
        'f1': f1,
        'precision': precision[idx],
        'recall': recall[idx],
        'breakeven': breakeven
    }

# Calculate metrics
